      function compare(a, b) {

        const dir = listState.sortDir === "asc" ? 1 : -1;

        const key = listState.sortKey;


        const av = valueForSort(a, key);

        const bv = valueForSort(b, key);


        if (av < bv) return -1 * dir;

        if (av > bv) return 1 * dir;

        return 0;

      }


      function valueForSort(s, key) {

        if (key === "dateISO") return s.dateISO || "";

        if (key === "type") return s.type || "";

        if (key === "description") return (s.description || "").toLowerCase();

        if (key === "locationLabel") return locationLabel(s).toLowerCase();

        if (key === "technicianName") return techName(s.technicianId).toLowerCase();

        if (key === "status") return s.status || "";

        if (key === "hours") return Number(s.hours) || 0;

        return "";

      }


      function applyListFilters(items) {

        const q = norm(listState.filters.q);

        const status = listState.filters.status;

        const techId = listState.filters.techId;

        const from = listState.filters.from;

        const to = listState.filters.to;


        return items.filter(s => {

          if (status && s.status !== status) return false;

          if (techId && s.technicianId !== techId) return false;

          if (from && s.dateISO < from) return false;

          if (to && s.dateISO > to) return false;


          if (q) {

            const hay = [

              s.description,

              locationLabel(s),

              techName(s.technicianId),

              s.type,

              s.status

            ].map(norm).join(" | ");

            if (!hay.includes(q)) return false;

          }

          return true;

        });

      }

      function countBy(items, keyFn) {

        const map = new Map();

        for (const it of items) {

          const k = keyFn(it) || "—";

          map.set(k, (map.get(k) || 0) + 1);

        }

        const labels = Array.from(map.keys());

        const values = labels.map(l => map.get(l));

        return { labels, values };

      }


      function buildLast12MonthsCounts(items) {

        const now = new Date();

        const labels = [];

        const keys = [];


        for (let i = 11; i >= 0; i--) {

          const d = new Date(now.getFullYear(), now.getMonth() - i, 1);

          const y = d.getFullYear();

          const m = String(d.getMonth() + 1).padStart(2, "0");

          const k = `${y}-${m}`;

          keys.push(k);

          labels.push(k);

        }


        const counts = Object.fromEntries(keys.map(k => [k, 0]));

        for (const s of items) {

          const k = (s.dateISO || "").slice(0, 7);

          if (counts[k] != null) counts[k] += 1;

        }


        return { labels, values: keys.map(k => counts[k]) };

      }

      function drawBarChart(canvas, labels, values, opts = {}) {

        const ctx = clearCanvas(canvas);

        const w = canvas.width, h = canvas.height;

        const pad = 34;

        const max = Math.max(1, ...values);


        // axes

        ctx.strokeStyle = "rgba(148,163,184,.55)";

        ctx.lineWidth = 1;

        ctx.beginPath();

        ctx.moveTo(pad, 10);

        ctx.lineTo(pad, h - pad);

        ctx.lineTo(w - 10, h - pad);

        ctx.stroke();


        const barAreaW = (w - pad - 20);

        const barW = Math.max(10, Math.floor(barAreaW / Math.max(1, values.length)) - 6);


        ctx.fillStyle = opts.color || "#1e40af";


        for (let i = 0; i < values.length; i++) {

          const v = values[i] || 0;

          const x = pad + 8 + i * (barW + 6);

          const barH = Math.round((v / max) * (h - pad - 26));

          const y = (h - pad) - barH;


          ctx.globalAlpha = 0.95;

          ctx.fillRect(x, y, barW, barH);


          // labels (simple)

          ctx.globalAlpha = 0.9;

          ctx.fillStyle = "rgba(15,23,42,.85)";

          if (document.documentElement.getAttribute("data-theme") === "dark") ctx.fillStyle = "rgba(248,250,252,.88)";

          ctx.font = "10px system-ui";

          ctx.fillText(String(v), x, y - 6);


          ctx.fillStyle = opts.color || "#1e40af";

        }


        // x labels (every 2 to avoid clutter)

        ctx.fillStyle = document.documentElement.getAttribute("data-theme") === "dark" ? "rgba(248,250,252,.75)" : "rgba(15,23,42,.70)";

        ctx.font = "10px system-ui";

        labels.forEach((lab, i) => {

          if (labels.length > 8 && i % 2 !== 0) return;

          const x = pad + 8 + i * (barW + 6);

          ctx.fillText(lab.slice(2), x, h - 14); // show YY-MM

        });

      }


      function drawPieChart(canvas, labels, values, colors) {

        const ctx = clearCanvas(canvas);

        const w = canvas.width, h = canvas.height;

        const cx = 130, cy = Math.floor(h / 2);

        const r = Math.min(90, Math.floor(h / 2) - 18);


        const total = values.reduce((a, b) => a + b, 0) || 1;

        let angle = -Math.PI / 2;


        for (let i = 0; i < values.length; i++) {

          const v = values[i] || 0;

          const slice = (v / total) * Math.PI * 2;


          ctx.beginPath();

          ctx.moveTo(cx, cy);

          ctx.arc(cx, cy, r, angle, angle + slice);

          ctx.closePath();

          ctx.fillStyle = colors[i % colors.length];

          ctx.fill();


          angle += slice;

        }


        // legend

        ctx.font = "12px system-ui";

        const startX = 260;

        let y = 34;

        for (let i = 0; i < labels.length; i++) {

          ctx.fillStyle = colors[i % colors.length];

          ctx.fillRect(startX, y - 10, 12, 12);


          ctx.fillStyle = document.documentElement.getAttribute("data-theme") === "dark" ? "rgba(248,250,252,.88)" : "rgba(15,23,42,.88)";

          const pct = Math.round(((values[i] || 0) / total) * 100);

          ctx.fillText(`${labels[i]} — ${values[i]} (${pct}%)`, startX + 18, y);

          y += 22;

        }

      }

      function validateImported(obj) {

        if (!obj || typeof obj !== "object") return { ok: false, reason: "Arquivo inválido." };

        if (!Array.isArray(obj.services) || !Array.isArray(obj.techs) || !Array.isArray(obj.equips)) {

          return { ok: false, reason: "Schema inválido: esperado {services, techs, equips, settings}." };

        }


        // minimal schema validation for services

        for (const s of obj.services) {

          if (!s || typeof s !== "object") return { ok: false, reason: "Schema inválido em services." };

          const must = ["id","dateISO","type","description","equipmentId","technicianId","status","hours","createdAtISO","updatedAtISO"];

          for (const k of must) {

            if (!(k in s)) return { ok: false, reason: `Campo ausente em serviço: ${k}` };

          }

          if (typeof s.dateISO !== "string" || s.dateISO.length !== 10) return { ok: false, reason: "dateISO inválido (use YYYY-MM-DD)." };

          if (!["Preventiva","Corretiva","Emergencial"].includes(s.type)) return { ok: false, reason: "type inválido em algum serviço." };

          if (!["Pendente","Em Andamento","Concluído"].includes(s.status)) return { ok: false, reason: "status inválido em algum serviço." };

          if (typeof s.description !== "string" || !s.description.trim()) return { ok: false, reason: "description inválida em algum serviço." };

          if (Number.isNaN(Number(s.hours)) || Number(s.hours) < 0) return { ok: false, reason: "hours inválido em algum serviço." };

        }


        // unique names validation (soft)

        const techNames = new Set();

        for (const t of obj.techs) {

          const n = norm(t?.name);

          if (!n) return { ok: false, reason: "Técnico sem nome no backup." };

          if (techNames.has(n)) return { ok: false, reason: "Backup contém técnicos com nomes duplicados." };

          techNames.add(n);

        }


        const equipNames = new Set();

        for (const e of obj.equips) {

          const n = norm(e?.name);

          if (!n) return { ok: false, reason: "Equipamento/local sem nome no backup." };

          if (equipNames.has(n)) return { ok: false, reason: "Backup contém equipamentos/locais com nomes duplicados." };

          equipNames.add(n);

        }


        return { ok: true };

      }


      async function importJSON() {

        const file = el.importFile.files?.[0];

        if (!file) return showToast("Selecione um arquivo JSON para importar.", "error");


        const text = await file.text();

        let obj = null;


        try {

          obj = JSON.parse(text);

        } catch {

          return showToast("JSON inválido.", "error");

        }


        const v = validateImported(obj);

        if (!v.ok) return showToast(v.reason, "error");


        askConfirm(

          `<p>Importar backup e <strong>sobrescrever</strong> os dados atuais?</p><p class="muted small">Esta ação substitui services/techs/equips/settings.</p>`,

          () => {

            services = obj.services;

            techs = obj.techs;

            equips = obj.equips;

            settings = obj.settings || { darkMode: false };


            save(LS.services, services);

            save(LS.techs, techs);

            save(LS.equips, equips);

            save(LS.settings, settings);


            applyTheme();

            refreshAll();

            showToast("Backup importado.", "success");

          }

        );

      }
